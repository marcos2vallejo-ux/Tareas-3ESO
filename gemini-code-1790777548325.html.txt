    <script>
        const originalClassmates = ["Asier", "Jorge", "Adriana", "Tamara", "Paola", "Sergio", "Izan", "Lorenzo", "Vallejo", "Ania", "Marxos"];
        let classmates = [...originalClassmates].sort();

        const selectEl = document.getElementById('user-select');
        classmates.forEach(name => {
            const opt = document.createElement('option');
            opt.value = name;
            opt.textContent = name;
            selectEl.appendChild(opt);
        });

        window.onload = function() {
            const savedUser = localStorage.getItem('santo_tomas_user');
            if (savedUser) {
                document.getElementById('login-screen').classList.add('hidden');
                document.getElementById('logged-user-badge').textContent = savedUser;
            }
            initApp();
            checkUserPermissions();
        };

        function saveUser() {
            const chosen = selectEl.value;
            if (!chosen) {
                alert("Por favor, selecciona tu nombre.");
                return;
            }
            localStorage.setItem('santo_tomas_user', chosen);
            document.getElementById('login-screen').classList.add('hidden');
            document.getElementById('logged-user-badge').textContent = chosen;
            checkUserPermissions();
        }

        function changeUser() {
            localStorage.removeItem('santo_tomas_user');
            document.getElementById('login-screen').classList.remove('hidden');
        }

        const schoolSchedule = {
            1: ["Biología", "Geografía", "Ed. Física", "Matemáticas", "Física", "Iniciativa / Robótica / Francés"],
            2: ["Matemáticas", "Plástica / Música", "Física", "Lengua", "Geografía", "Inglés"],
            3: ["Matemáticas", "Ed. Física", "Plástica / Música", "Religión / ARE", "Lengua", "Inglés"],
            4: ["Tecnología", "Inglés", "Lengua", "Plástica / Música", "Tutoría", "Valores Éticos"],
            5: ["Tecnología", "Geografía", "Matemáticas", "Lengua", "Iniciativa / Robótica / Francés", "Biología"]
        };

        function getResponsableToday() {
            const now = new Date();
            let schoolDayCounter = 0;
            let d = new Date(now.getFullYear(), 0, 1);
            while (d <= now) {
                let dayOfWeek = d.getDay();
                if (dayOfWeek >= 1 && dayOfWeek <= 5) {
                    schoolDayCounter++;
                }
                d.setDate(d.getDate() + 1);
            }

            let currentDayOfWeek = now.getDay();
            if (currentDayOfWeek === 0 || currentDayOfWeek === 6) {
                schoolDayCounter--;
            }

            const index = (schoolDayCounter - 1) % originalClassmates.length;
            const safeIndex = index < 0 ? 0 : index;
            const nextIndex = (safeIndex + 1) % originalClassmates.length;

            return {
                current: originalClassmates[safeIndex],
                currentIndex: safeIndex,
                next: originalClassmates[nextIndex]
            };
        }

        const respInfo = getResponsableToday();
        document.getElementById('today-responsable-name').textContent = respInfo.current;
        document.getElementById('next-responsable-name').textContent = respInfo.next;

        const now = new Date();
        document.getElementById('day-of-week-label').textContent = now.toLocaleDateString('es-ES', { weekday: 'long', day: 'numeric', month: 'long' });

        let jsDay = now.getDay();
        let todaySubjects = schoolSchedule[jsDay] || schoolSchedule[5];

        function initApp() {
            const container = document.getElementById('schedule-subjects-container');
            container.innerHTML = '';
            todaySubjects.forEach((subj, idx) => {
                const div = document.createElement('div');
                div.className = "bg-slate-50 border border-slate-200 rounded-xl p-3 space-y-2";
                div.innerHTML = `
                    <div class="flex justify-between items-center">
                        <span class="font-bold text-indigo-900 text-xs">📖 ${subj}</span>
                        <label class="flex items-center gap-1.5 text-xs text-slate-600 bg-white px-2 py-1 rounded-md border border-slate-200 cursor-pointer">
                            <input type="checkbox" id="no-task-${idx}" onchange="toggleNoTask(${idx})" class="rounded text-indigo-600 focus:ring-indigo-500 w-3.5 h-3.5">
                            <span>Sin tarea</span>
                        </label>
                    </div>
                    <div id="wrapper-${idx}" class="space-y-2">
                        <textarea id="desc-${idx}" rows="2" placeholder="¿Qué hay que hacer en ${subj}?" class="w-full text-xs p-2.5 border border-slate-300 rounded-lg focus:ring-1 focus:ring-indigo-500 focus:outline-none bg-white"></textarea>
                        <div class="space-y-1">
                            <input type="url" id="link-${idx}" placeholder="🔗 Enlace Google Doc (Opcional)" class="w-full text-xs p-2 border border-slate-300 rounded-lg focus:ring-1 focus:ring-indigo-500 focus:outline-none bg-white">
                            <div class="flex items-center gap-2 pt-1">
                                <label class="w-full bg-white border border-slate-300 text-slate-600 text-xs py-2 px-3 rounded-lg flex items-center justify-center gap-1.5 cursor-pointer active:bg-slate-50">
                                    <span>📷 Subir foto de la galería</span>
                                    <input type="file" id="img-file-${idx}" accept="image/*" class="hidden" onchange="handleImage(this, ${idx})">
                                </label>
                            </div>
                            <input type="hidden" id="img-data-${idx}">
                            <p id="img-status-${idx}" class="text-[10px] text-emerald-600 font-medium hidden">✅ Imagen cargada correctamente</p>
                        </div>
                    </div>
                `;
                container.appendChild(div);
            });

            const classmatesContainer = document.getElementById('classmates-list');
            classmatesContainer.innerHTML = '';
            originalClassmates.forEach((name, i) => {
                const isToday = i === respInfo.currentIndex;
                const row = document.createElement('div');
                row.className = `py-2.5 px-2 flex justify-between items-center text-xs ${isToday ? 'bg-indigo-50 font-bold text-indigo-700 rounded-lg' : 'text-slate-600'}`;
                row.innerHTML = `
                    <span>${i + 1}. ${name}</span>
                    ${isToday ? '<span class="text-[10px] bg-indigo-600 text-white px-2 py-0.5 rounded-full">¡Hoy!</span>' : ''}
                `;
                classmatesContainer.appendChild(row);
            });

            renderHistory();
        }

        function handleImage(input, idx) {
            const file = input.files[0];
            if (file) {
                const reader = new FileReader();
                reader.onload = function(e) {
                    document.getElementById(`img-data-${idx}`).value = e.target.result;
                    document.getElementById(`img-status-${idx}`).classList.remove('hidden');
                };
                reader.readAsDataURL(file);
            }
        }

        function checkUserPermissions() {
            const currentUser = localStorage.getItem('santo_tomas_user');
            const isResponsible = currentUser === respInfo.current;
            const warningBox = document.getElementById('permission-warning');
            const warningText = document.getElementById('warning-text');
            const submitBtn = document.getElementById('submit-btn');

            if (!isResponsible) {
                warningBox.classList.remove('hidden');
                warningText.textContent = `Hoy le toca subir las tareas a ${respInfo.current}. Puedes ver los campos, pero están bloqueados para edición.`;
                submitBtn.disabled = true;
                submitBtn.classList.add('opacity-50', 'cursor-not-allowed');
                
                document.querySelectorAll('#schedule-subjects-container input, #schedule-subjects-container textarea').forEach(el => {
                    el.disabled = true;
                });
            } else {
                warningBox.classList.add('hidden');
                submitBtn.disabled = false;
                submitBtn.classList.remove('opacity-50', 'cursor-not-allowed');
                
                document.querySelectorAll('#schedule-subjects-container input, #schedule-subjects-container textarea').forEach(el => {
                    el.disabled = false;
                });
            }
        }

        function toggleNoTask(idx) {
            const chk = document.getElementById(`no-task-${idx}`);
            const wrap = document.getElementById(`wrapper-${idx}`);
            if (chk.checked) {
                wrap.style.opacity = "0.4";
                wrap.querySelectorAll('input, textarea').forEach(el => el.disabled = true);
            } else {
                wrap.style.opacity = "1";
                wrap.querySelectorAll('input, textarea').forEach(el => el.disabled = false);
            }
        }

        let tasksHistory = JSON.parse(localStorage.getItem('clase_santo_tomas_tasks')) || [];

        function renderHistory() {
            const histContainer = document.getElementById('history-list');
            histContainer.innerHTML = '';
            if (tasksHistory.length === 0) {
                histContainer.innerHTML = '<p class="text-slate-400 text-center py-6 text-xs">No hay tareas publicadas todavía.</p>';
                return;
            }

            tasksHistory.forEach(item => {
                let subsHtml = '';
                item.subjects.forEach(s => {
                    subsHtml += `
                        <div class="bg-slate-50 p-2.5 rounded-lg border border-slate-100 text-xs space-y-1">
                            <div class="flex justify-between font-semibold text-slate-700">
                                <span>${s.name}</span>
                                <span class="${s.noTask ? 'text-emerald-600' : 'text-slate-400'}">${s.noTask ? '✅ Sin tarea' : ''}</span>
                            </div>
                            ${!s.noTask ? `<p class="text-slate-600">${s.desc || 'Sin descripción detallada.'}</p>` : ''}
                            <div class="flex flex-col gap-2 pt-1">
                                ${s.link ? `<a href="${s.link}" target="_blank" class="text-[11px] text-indigo-600 underline font-medium">🔗 Ver enlace de Google Doc</a>` : ''}
                                ${s.imgData ? `<a href="${s.imgData}" target="_blank"><img src="${s.imgData}" class="rounded-lg max-h-32 object-cover border border-slate-200 mt-1"></a>` : ''}
                            </div>
                        </div>
                    `;
                });

                const card = document.createElement('div');
                card.className = "bg-white rounded-2xl p-4 shadow-sm border border-slate-200 space-y-2";
                card.innerHTML = `
                    <div class="flex justify-between items-center border-b border-slate-100 pb-2">
                        <span class="font-bold text-indigo-700 text-xs">📅 ${item.dateStr}</span>
                        <span class="text-xs bg-slate-100 text-slate-600 px-2.5 py-0.5 rounded-md font-medium">Subido por: ${item.author}</span>
                    </div>
                    <div class="space-y-2 pt-1">
                        ${subsHtml}
                    </div>
                `;
                histContainer.appendChild(card);
            });
        }

        function switchTab(tabName) {
            ['today', 'history', 'rotacion'].forEach(t => {
                document.getElementById(`tab-${t}`).classList.add('hidden');
                document.getElementById(`nav-btn-${t}`).className = "flex flex-col items-center text-slate-400 hover:text-slate-600 transition-colors";
            });
            document.getElementById(`tab-${tabName}`).classList.remove('hidden');
            document.getElementById(`nav-btn-${tabName}`).className = "flex flex-col items-center text-indigo-600 transition-colors";
            window.scrollTo(0, 0);
        }

        document.getElementById('task-form').addEventListener('submit', (e) => {
            e.preventDefault();
            const currentUser = localStorage.getItem('santo_tomas_user');
            
            if (currentUser !== respInfo.current) {
                alert("¡Solo el responsable de hoy puede publicar o modificar las tareas!");
                return;
            }
            
            let subjectsData = [];
            todaySubjects.forEach((subj, idx) => {
                const noTask = document.getElementById(`no-task-${idx}`).checked;
                const desc = document.getElementById(`desc-${idx}`).value;
                const link = document.getElementById(`link-${idx}`).value;
                const imgData = document.getElementById(`img-data-${idx}`).value;
                subjectsData.push({ name: subj, noTask, desc, link, imgData });
            });

            const entry = {
                dateStr: now.toLocaleDateString('es-ES', { weekday: 'short', day: 'numeric', month: 'short' }),
                author: currentUser,
                subjects: subjectsData
            };

            const todayStr = entry.dateStr;
            tasksHistory = tasksHistory.filter(item => item.dateStr !== todayStr);

            tasksHistory.unshift(entry);
            localStorage.setItem('clase_santo_tomas_tasks', JSON.stringify(tasksHistory));
            alert('¡Tareas guardadas y actualizadas correctamente!');
            renderHistory();
            switchTab('history');
        });
    </script>
